driver_counts <- drivers %>%
  filter(!is.na(total_experience)) %>%
  mutate(exp_group = cut(
    total_experience, 
    breaks = c(0, 5, 10, 15, 20, Inf), 
    labels = c("0-4 years", "5-9 years", "10-14 years", "15-19 years", "20+ years"),
    right = FALSE 
  )) %>%
  group_by(exp_group) %>%
  summarize(total_drivers = n())

incident_counts <- incidents %>%
  left_join(drivers, by = "driver_id") %>%
  filter(!is.na(total_experience)) %>%
  mutate(exp_group = cut(
    total_experience, 
    breaks = c(0, 5, 10, 15, 20, Inf), 
    labels = c("0-4 years", "5-9 years", "10-14 years", "15-19 years", "20+ years"),
    right = FALSE
  )) %>%
  group_by(exp_group) %>%
  summarize(total_incidents = n())

incident_analysis_grouped <- driver_counts %>%
